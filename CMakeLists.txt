cmake_minimum_required(VERSION 3.15)
project(SITRAC LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(MODEL src/interfaz.cpp src/util.cpp src/ciudades.cpp src/rutas.cpp src/pasajeros.cpp src/cola.cpp src/historial.cpp src/busquedas.cpp src/reportes.cpp src/archivos.cpp)
add_executable(SITRAC ${MODEL} src/windows.cpp windows.rc)
target_link_libraries(SITRAC PRIVATE comctl32 comdlg32)
if(MSVC)
  target_compile_options(SITRAC PRIVATE /W4)
else()
  target_compile_options(SITRAC PRIVATE -Wall -Wextra -Werror)
endif()
